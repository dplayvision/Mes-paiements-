Calcul des mois écoulés depuis la date de début jusqu’à aujourd'hui let yearDiff = today.getFullYear() - start.getFullYear(); let monthDiff = today.getMonth() - start.getMonth(); let totalMonths = yearDiff * 12 + monthDiff; // Si le jour de prélèvement du mois en cours est déjà passé ou atteint, on compte le mois actuel if (today.getDate() >= day) { totalMonths += 1; } const count = Math.max(0, totalMonths); const total = count * amount; // 2. Calcul du prochain prélèvement let nextYear = today.getFullYear(); let nextMonth = today.getMonth(); // 0-indexé if (today.getDate() >= day) { nextMonth += 1; // Passe au mois suivant if (nextMonth > 11) { nextMonth = 0; nextYear += 1; } } // Sécurité pour les mois qui n'ont pas 31 jours (ex: 28 ou 30) const maxDaysInNextMonth = new Date(nextYear, nextMonth + 1, 0).getDate(); const actualDay = Math.min(day, maxDaysInNextMonth); const nextDateObj = new Date(nextYear, nextMonth, actualDay); // Format DD/MM/YYYY const d = String(nextDateObj.getDate()).padStart(2, '0'); const m = String(nextDateObj.getMonth() + 1).padStart(2, '0'); const y = nextDateObj.getFullYear(); const nextDateStr = `${d}/${m}/${y}`; return { count: count, total: total, nextDate: nextDateStr }; } function renderPermanentDomiciliations() { const list = document.getElementById("permanentList"); if (!list) return; list.innerHTML = ""; if (!permanentDomiciliations || permanentDomiciliations.length === 0) { list.innerHTML = `
Aucune domiciliation permanente.

• Prochain prélèvement : ${stats.nextDate}
• Total prélevé : ${stats.total.toFixed(2).replace('.', ',')} €
• ${stats.count} prélèvements effectués
`; list.appendChild(item); }); } function deletePermanent(id) { if (!confirm("Voulez-vous vraiment supprimer cette domiciliation permanente ?")) return; permanentDomiciliations = permanentDomiciliations.filter(p => String(p.id) !== String(id)); savePermanentDomiciliations(); renderPermanentDomiciliations(); } /* ================================================== INTEGRATION HTML À PLACER DANS LA PAGE ================================================== */ /* Collez cette structure HTML à l'endroit souhaité dans votre interface :
